import assert from "node:assert/strict";
import { model } from "../../../models/city-land-use/model.ts";
const instance = model.create({ residentialDemand: 12, commercialDemand: 0 });
instance.setProtection("parcel:6", true);
instance.advance(10);
let state = instance.read();
assert.equal(state.residentialAreaHa, 9);
assert.equal(state.commercialAreaHa, 0);
assert.equal(state.greenAreaHa, 3);
assert.equal(state.unmetResidentialDemand, 3);
assert.equal(state.parcels[6]?.use, "green");
const historyLength = state.history.length;
instance.setProtection("parcel:6", false);
assert.equal(instance.read().tick, 10);
assert.equal(instance.read().history.length, historyLength);
instance.advance();
assert.equal(instance.read().residentialAreaHa, 10);
instance.setDemand({ residentialDemand: 0, commercialDemand: 0 });
instance.advance(5);
state = instance.read();
assert.equal(state.residentialAreaHa, 0);
assert.equal(state.greenAreaHa, 12);
assert.equal(state.protectedAreaHa, 2);
assert.equal(state.totalAreaHa, 12);
assert.equal(state.unmetCommercialDemand, 0);
console.log("city-land-use policy passed");
