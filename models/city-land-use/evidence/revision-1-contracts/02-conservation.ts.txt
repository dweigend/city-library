import assert from "node:assert/strict";
import { model } from "../../../models/city-land-use/model.ts";
const instance = model.create();
const counts = () => { const state = instance.read(); return [state.residentialAreaHa, state.commercialAreaHa, state.greenAreaHa]; };
assert.deepEqual(counts(), [4, 2, 6]);
instance.advance();
assert.deepEqual(counts(), [6, 2, 4]);
assert.equal(instance.read().convertedParcelCount, 2);
assert.equal(instance.read().parcels[6]?.use, "residential");
assert.equal(instance.read().parcels[7]?.use, "residential");
instance.advance();
assert.deepEqual(counts(), [6, 3, 3]);
assert.equal(instance.read().parcels[8]?.use, "commercial");
assert.equal(instance.read().convertedParcelCount, 1);
instance.advance(200);
const state = instance.read();
assert.equal(state.tick, 202);
assert.deepEqual(counts(), [6, 3, 3]);
assert.equal(state.convertedParcelCount, 0);
assert.equal(state.history.length, 121);
assert.equal(state.history[0]?.tick, 82);
for (const sample of state.history) assert.equal(sample.values.residentialAreaHa! + sample.values.commercialAreaHa! + sample.values.greenAreaHa!, 12);
for (const parcel of state.parcels) assert.equal(parcel.areaHa, 1);
assert.equal(state.totalAreaHa, 12);
console.log("city-land-use conservation passed");
