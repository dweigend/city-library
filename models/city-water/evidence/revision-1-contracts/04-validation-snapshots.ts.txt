import assert from "node:assert/strict";
import { model } from "../../../models/city-water/model.ts";
assert.throws(() => model.create({ initialStoredM3: 1001 }), RangeError);
assert.throws(() => model.create({ inflowM3PerDay: NaN }), RangeError);
assert.throws(() => model.create({ conservationShare: 1.1 }), RangeError);
assert.throws(() => model.create({ wastewaterShare: -0.1 }), RangeError);
// @ts-expect-error Unknown options must also fail at runtime.
assert.throws(() => model.create({ seed: 1 }), TypeError);
const instance = model.create();
const snapshot = instance.read();
assert.throws(() => instance.advance(-1), RangeError);
assert.throws(() => instance.advance(0.5), RangeError);
assert.throws(() => instance.setConservationShare(NaN), RangeError);
assert.throws(() => instance.setTreatmentCapacity(-1), RangeError);
assert.throws(() => instance.setInflow(Infinity), RangeError);
assert.deepEqual(instance.read(), snapshot);
Reflect.set(snapshot, "storedM3", -99);
Reflect.set(snapshot.parameters, "conservationShare", 1);
Reflect.set(snapshot.history[0]!.values, "storedM3", -99);
assert.equal(instance.read().storedM3, 600);
assert.equal(instance.read().parameters.conservationShare, 0);
assert.equal(instance.read().history[0]!.values.storedM3, 600);
const other = model.create();
instance.advance(10);
other.advance(10);
assert.deepEqual(instance.read(), other.read());
assert.equal(snapshot.storedM3, 600);
assert.equal(snapshot.history.length, 1);
instance.advance();
assert.equal(other.read().tick, 10);
instance.advance(400);
assert.equal(instance.read().history.length, 366);
assert.equal(instance.read().history.at(-1)?.tick, 411);
assert.ok(Math.abs(instance.read().balanceErrorM3) < 1e-9);
console.log("city-water validation, determinism and snapshots passed");
