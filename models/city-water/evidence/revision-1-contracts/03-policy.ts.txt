import assert from "node:assert/strict";
import { model } from "../../../models/city-water/model.ts";
const instance = model.create();
instance.advance();
const before = instance.read();
instance.setConservationShare(0.5);
instance.setTreatmentCapacity(200);
instance.setInflow(0);
assert.equal(instance.read().tick, 1);
assert.equal(instance.read().storedM3, before.storedM3);
assert.deepEqual(instance.read().history, before.history);
instance.advance();
const state = instance.read();
assert.equal(state.demandM3, 100);
assert.equal(state.suppliedM3, 100);
assert.equal(state.storedM3, 400);
assert.equal(state.wastewaterM3, 80);
assert.equal(state.treatedM3, 80);
assert.equal(state.untreatedM3, 0);
assert.equal(state.totalInflowM3, 100);
assert.equal(state.totalSuppliedM3, 300);
assert.equal(state.totalTreatedM3, 200);
assert.equal(state.balanceErrorM3, 0);
console.log("city-water policy intervention passed");
