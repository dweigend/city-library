import assert from "node:assert/strict";
import { model } from "../../../models/city-water/model.ts";
const instance = model.create();
instance.advance();
let state = instance.read();
assert.equal(state.tick, 1);
assert.equal(state.storedM3, 500);
assert.equal(state.demandM3, 200);
assert.equal(state.suppliedM3, 200);
assert.equal(state.shortageM3, 0);
assert.equal(state.wastewaterM3, 160);
assert.equal(state.treatedM3, 120);
assert.equal(state.untreatedM3, 40);
assert.equal(state.consumedM3, 40);
instance.advance();
state = instance.read();
assert.equal(state.storedM3, 400);
assert.equal(state.totalInflowM3, 200);
assert.equal(state.totalSuppliedM3, 400);
assert.equal(state.totalTreatedM3, 240);
assert.equal(state.totalUntreatedM3, 80);
assert.equal(state.totalConsumedM3, 80);
assert.equal(state.balanceErrorM3, 0);
const drought = model.create({ initialStoredM3: 50, inflowM3PerDay: 0 });
drought.advance();
assert.equal(drought.read().storedM3, 0);
assert.equal(drought.read().suppliedM3, 50);
assert.equal(drought.read().shortageM3, 150);
assert.equal(drought.read().wastewaterM3, 40);
assert.equal(drought.read().consumedM3, 10);
assert.equal(drought.read().balanceErrorM3, 0);
const spill = model.create({ initialStoredM3: 1000, inflowM3PerDay: 300 });
spill.advance();
assert.equal(spill.read().spillM3, 300);
assert.equal(spill.read().storedM3, 800);
assert.equal(spill.read().balanceErrorM3, 0);
console.log("city-water first steps and conservation passed");
