import assert from "node:assert/strict";
import { model } from "../../../models/city-daily-activities/model.ts";
assert.throws(() => model.create({ householdCount: 0 }), RangeError);
assert.throws(() => model.create({ householdCount: 1.5 }), RangeError);
assert.throws(() => model.create({ remoteWork: 1 } as never), TypeError);
assert.throws(() => model.create({ unknown: 1 } as never), TypeError);
const first = model.create({ householdCount: 1 });
const second = model.create({ householdCount: 1 });
const initial = first.read();
assert.throws(() => first.advance(-1), RangeError);
assert.throws(() => first.advance(0.5), RangeError);
assert.throws(() => first.setRemoteWork("yes" as never), TypeError);
assert.deepEqual(first.read(), initial);
first.advance(48);
second.advance(48);
assert.deepEqual(first.read(), second.read());
assert.equal(initial.tick, 0);
assert.equal(initial.residents[0]?.location, "home");
assert.notEqual(first.read().residents, first.read().residents);
assert.notEqual(first.read().residents[0], first.read().residents[0]);
assert.notEqual(first.read().personHours, first.read().personHours);
const saved = second.read();
first.setRemoteWork(true);
first.advance();
assert.deepEqual(second.read(), saved);
second.advance(0);
assert.deepEqual(second.read(), saved);
console.log("daily activities validation and snapshots passed");
