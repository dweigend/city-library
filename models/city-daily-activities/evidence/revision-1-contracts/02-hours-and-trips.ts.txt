import assert from "node:assert/strict";
import { model } from "../../../models/city-daily-activities/model.ts";
const instance = model.create({ householdCount: 1 });
instance.advance();
assert.deepEqual(instance.read().personHours, { home: 2, work: 0, school: 0, services: 0, travel: 0 });
instance.advance(7);
assert.equal(instance.read().tripCount, 1);
assert.equal(instance.read().atSchoolCount, 1);
instance.advance();
assert.deepEqual(instance.read().personHours, { home: 15, work: 0, school: 1, services: 0, travel: 2 });
instance.advance(15);
const state = instance.read();
assert.equal(state.tick, 24);
assert.equal(state.hourOfDay, 0);
assert.equal(state.tripCount, 5);
assert.equal(state.atHomeCount, 2);
assert.equal(state.accountedPersonHours, 48);
assert.deepEqual(state.personHours, { home: 29, work: 7, school: 6, services: 1, travel: 5 });
assert.equal(Object.values(state.personHours).reduce((sum, value) => sum + value, 0), state.residentCount * state.tick);
console.log("daily activities hourly conservation passed");
