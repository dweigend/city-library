import assert from "node:assert/strict";
import { model } from "../../../models/city-daily-activities/model.ts";
const instance = model.create({ householdCount: 1 });
instance.advance(8);
const before = instance.read();
instance.setRemoteWork(true);
assert.equal(instance.read().tick, before.tick);
assert.equal(instance.read().tripCount, before.tripCount);
assert.equal(instance.read().parameters.remoteWork, true);
instance.advance(16);
assert.equal(instance.read().tripCount, 4);
assert.deepEqual(instance.read().personHours, { home: 37, work: 0, school: 6, services: 1, travel: 4 });
instance.setRemoteWork(false);
instance.advance(24);
assert.equal(instance.read().tripCount, 9);
assert.equal(instance.read().accountedPersonHours, 96);
console.log("daily activities live policy passed");
