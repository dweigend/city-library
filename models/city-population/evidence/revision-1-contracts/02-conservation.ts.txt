import assert from "node:assert/strict";
import { model } from "../../../models/city-population/model.ts";

const instance = model.create();
instance.advance();
const first = instance.read();
assert.equal(first.tick, 1);
assert.equal(first.birthCount, 20);
assert.equal(first.deathCount, 100);
assert.equal(first.netMigration, 5);
assert.equal(first.populationCount, 925);
assert.equal(first.cohorts[0]!.residentCount, 20);
assert.equal(first.cohorts[25]!.residentCount, 14);
assert.equal(first.cohorts[100]!.residentCount, 9);
assert.equal(first.householdCount, 309);
assert.equal(first.dissolvedHouseholdCount, 25);
instance.advance();
const second = instance.read();
assert.equal(second.populationCount, 945);
assert.equal(second.birthCount, 18);
assert.equal(second.deathCount, 3);
assert.equal(second.cohorts[1]!.residentCount, 18);
assert.equal(second.cohorts[100]!.residentCount, 18);
assert.equal(second.formedHouseholdCount, 6);
for (let year = 0; year < 200; year += 1) {
  const previous = instance.read();
  instance.advance();
  const state = instance.read();
  assert.equal(state.populationCount, previous.populationCount + state.birthCount - state.deathCount + state.netMigration);
  assert.equal(state.populationCount, state.cohorts.reduce((sum, cohort) => sum + cohort.residentCount, 0));
  assert.equal(state.populationCount, state.householdsBySize.reduce((sum, group) => sum + group.size * group.householdCount, 0));
  assert.equal(state.accountingError, 0);
}
console.log("city-population first steps and conservation passed");
