import assert from "node:assert/strict";
import { model } from "../../../models/city-finance/model.ts";

const instance = model.create();
instance.setPolicy({ maintenanceBudget: 400 });
assert.equal(instance.read().tick, 0);
assert.equal(instance.read().parameters.maintenanceBudget, 400);
instance.advance();
assert.equal(instance.read().maintenanceSpending, 300);
assert.equal(instance.read().cash, 0);
assert.equal(instance.read().backlog, 1900);
assert.equal(instance.read().condition, 0.81);
instance.setPolicy({ taxRate: 0.12 });
instance.advance();
assert.equal(instance.read().taxRevenue, 1200);
assert.equal(instance.read().maintenanceSpending, 400);
assert.equal(instance.read().cash, 0);
assert.equal(instance.read().backlog, 1700);
assert.equal(instance.read().condition, 0.83);
const restored = model.create({ initialBacklog: 0, annualWear: 200, maintenanceBudget: 1000, initialCash: 1000 });
restored.advance();
assert.equal(restored.read().maintenanceSpending, 200);
assert.equal(restored.read().backlog, 0);
assert.equal(restored.read().condition, 1);
assert.equal(restored.read().cash, 1000);
const worn = model.create({ taxRate: 0, maintenanceBudget: 0, initialBacklog: 9900 });
worn.advance(2);
assert.equal(worn.read().backlog, 10000);
assert.equal(worn.read().condition, 0);
assert.equal(worn.read().cumulativeWear, 100);
console.log("city-finance policy passed");
