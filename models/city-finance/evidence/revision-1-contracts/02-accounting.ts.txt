import assert from "node:assert/strict";
import { model } from "../../../models/city-finance/model.ts";

const instance = model.create();
instance.advance();
let state = instance.read();
assert.equal(state.tick, 1);
assert.equal(state.taxRevenue, 1000);
assert.equal(state.serviceSpending, 800);
assert.equal(state.maintenanceSpending, 150);
assert.equal(state.wear, 200);
assert.equal(state.cash, 150);
assert.equal(state.backlog, 2050);
assert.equal(state.condition, 0.795);
instance.advance();
state = instance.read();
assert.equal(state.cash, 200);
assert.equal(state.backlog, 2100);
assert.equal(state.condition, 0.79);
assert.equal(state.cumulativeTaxRevenue, 2000);
assert.equal(state.cumulativeServiceSpending, 1600);
assert.equal(state.cumulativeMaintenanceSpending, 300);
assert.equal(state.cash, 100 + state.cumulativeTaxRevenue - state.cumulativeServiceSpending - state.cumulativeMaintenanceSpending);
assert.equal(state.backlog, 2000 + state.cumulativeWear - state.cumulativeMaintenanceSpending);
const deficit = model.create({ initialCash: 0, taxRate: 0 });
deficit.advance();
assert.equal(deficit.read().cash, -800);
assert.equal(deficit.read().maintenanceSpending, 0);
assert.equal(deficit.read().backlog, 2200);
console.log("city-finance accounting passed");
