import assert from "node:assert/strict";
import { model } from "../../../models/city-air-noise/model.ts";

const instance = model.create();
instance.advance();
const before = instance.read();
instance.setTraffic(0);
const edited = instance.read();
assert.equal(edited.tick, 1);
assert.equal(edited.emissionMicrogramsPerMinute, 0);
assert.equal(edited.maximumNoiseDecibels, 40);
assert.deepEqual(edited.receptors.map(receptor => receptor.massMicrograms), before.receptors.map(receptor => receptor.massMicrograms));
assert.deepEqual(edited.history, before.history);
instance.advance();
const after = instance.read();
assert.deepEqual(after.receptors.map(receptor => receptor.massMicrograms), [281.25, 562.5, 281.25]);
assert.deepEqual(after.receptors.map(receptor => receptor.concentrationExposureMicrogramMinutesPerCubicMeter), [1.03125, 1.3125, 0.28125]);
assert.equal(after.emittedMassMicrograms, 2000);
assert.ok(Math.abs(after.receptors[0]!.noiseExposureJoulesPerSquareMeter - 0.0001212) < 1e-14);
instance.advance(100);
assert.ok(instance.read().retainedMassMicrograms < 1e-20);
const halfTraffic = model.create({ trafficVehiclesPerHour: 30 }).read();
assert.ok(Math.abs(halfTraffic.maximumNoiseDecibels - 10 * Math.log10(1010000)) < 1e-10);
console.log("city-air-noise policy: passed");
