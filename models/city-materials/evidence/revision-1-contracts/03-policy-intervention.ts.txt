import assert from "node:assert/strict";
import { model } from "../../../models/city-materials/model.ts";

const instance = model.create();
instance.advance(2);
const before = instance.read();
instance.setPolicy({ collectionCapacity: 140, reuseFraction: 0.5, recyclingFraction: 1 });
const changed = instance.read();
assert.equal(changed.tick, 2);
assert.deepEqual(changed.totals, before.totals);
assert.deepEqual(changed.flows, before.flows);
assert.deepEqual(changed.history, before.history);
instance.advance();
const state = instance.read();
assert.equal(state.uncollected, 20);
assert.deepEqual(state.flows, { generated: 100, collected: 140, reused: 70, recycled: 70, disposed: 0 });
assert.deepEqual(state.totals, { generated: 300, reused: 110, recycled: 130, disposed: 60 });
assert.equal(state.balanceError, 0);
assert.equal(before.parameters.collectionCapacity, 80);
console.log("city-materials policy passed");
