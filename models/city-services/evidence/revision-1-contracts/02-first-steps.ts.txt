import assert from "node:assert/strict";
import { model } from "../../../models/city-services/model.ts";

const instance = model.create();
instance.advance();
const first = instance.read();
assert.equal(first.tick, 1);
assert.equal(first.demandCount, 12);
assert.equal(first.servedCount, 6);
assert.equal(first.unmetCount, 6);
assert.ok(Math.abs(first.totalDistanceKm - 3 * (1 + Math.SQRT2)) < 1e-12);
assert.ok(Math.abs(first.meanDistanceKm - (1 + Math.SQRT2) / 2) < 1e-12);
for (const kind of ["school", "clinic", "shop"] as const) {
  const requests = first.requests.filter(request => request.kind === kind);
  assert.deepEqual(requests.map(request => request.facilityId), [`${kind}:west`, null, `${kind}:east`, null]);
  assert.deepEqual(requests.map(request => request.status), ["served", "capacity", "served", "capacity"]);
}
assert.ok(first.facilities.every(facility => facility.used === 1 && facility.capacity === 1));
instance.advance(2);
const third = instance.read();
assert.equal(third.tick, 3);
assert.equal(third.totalServedCount, 18);
assert.equal(third.totalUnmetCount, 18);
assert.equal(third.totalDemandCount, 36);
assert.equal(third.servedCount + third.unmetCount, third.demandCount);
assert.equal(third.totalServedCount + third.totalUnmetCount, third.totalDemandCount);
assert.deepEqual(third.requests, first.requests);
assert.deepEqual(third.history.map(sample => sample.tick), [0, 1, 2, 3]);
console.log("city-services first steps: passed");
