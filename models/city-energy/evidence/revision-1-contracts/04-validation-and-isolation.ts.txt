import assert from "node:assert/strict";
import { model } from "../../../models/city-energy/model.ts";
assert.throws(() => model.create({ batteryCapacityKwh: 3 }), RangeError);
for (const value of [-1, NaN, Infinity, 1001]) {
  assert.throws(() => model.create({ windOutputKw: value }), RangeError);
}
const instance = model.create();
const initial = instance.read();
for (const count of [-1, 0.5, NaN, 1_000_001]) assert.throws(() => instance.advance(count), RangeError);
assert.throws(() => instance.setNetworkCapacity({ electricityNetworkCapacityKw: 9, heatNetworkCapacityKw: -1 }), RangeError);
assert.deepEqual(instance.read(), initial);
const peer = model.create();
instance.advance(48);
peer.advance(24);
peer.advance(24);
assert.deepEqual(instance.read(), peer.read());
const snapshot = instance.read();
assert.notEqual(snapshot.flow, instance.read().flow);
assert.notEqual(snapshot.history, instance.read().history);
assert.notEqual(snapshot.parameters, instance.read().parameters);
assert.equal(Reflect.set(snapshot.flow, "renewableKwh", 999), false);
assert.equal(Reflect.set(snapshot.parameters, "demandScale", 99), false);
instance.advance(300);
assert.equal(snapshot.tick, 48);
assert.equal(peer.read().tick, 48);
assert.equal(initial.tick, 0);
assert.equal(initial.batteryKwh, 4);
assert.equal(instance.read().history.length, 169);
const zero = model.create({ batteryCapacityKwh: 0, initialBatteryKwh: 0, batteryPowerKw: 0,
  solarCapacityKw: 0, windOutputKw: 0, electricityNetworkCapacityKw: 0, heatNetworkCapacityKw: 0, demandScale: 0 });
zero.advance(24);
assert.equal(zero.read().totalElectricityUnmetKwh, 0);
assert.equal(zero.read().batteryKwh, 0);
console.log("city-energy validation and isolation passed");
