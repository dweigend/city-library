import assert from "node:assert/strict";
import { model } from "../../../models/city-energy/model.ts";
const instance = model.create();
const state = instance.read();
assert.equal(model.title, "Urban Energy Supply");
assert.equal(state.tick, 0);
assert.equal(state.intervalHour, null);
assert.equal(state.batteryKwh, 4);
assert.equal(state.history.length, 1);
assert.deepEqual(state.parameters, {
  solarCapacityKw: 12, windOutputKw: 2, batteryCapacityKwh: 8,
  initialBatteryKwh: 4, batteryPowerKw: 2, electricityNetworkCapacityKw: 6,
  heatNetworkCapacityKw: 3, demandScale: 1,
});
assert.equal(state.flow.electricityDemandKwh, 0);
assert.equal(state.totalElectricityUnmetKwh, 0);
instance.advance(0);
assert.deepEqual(instance.read(), state);
console.log("city-energy setup passed");
