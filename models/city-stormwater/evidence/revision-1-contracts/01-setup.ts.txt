import assert from "node:assert/strict";
import { model } from "../../../models/city-stormwater/model.ts";

const instance = model.create();
const state = instance.read();
assert.equal(model.title, "Urban Stormwater");
assert.deepEqual(state.parameters, {
  rainMmPerHour: 20, infiltrationMmPerHour: 10, retentionCapacityM3: 2, outletM3PerHour: 0.5,
});
assert.equal(state.tick, 0);
assert.equal(state.areaM2, 200);
assert.equal(state.storedM3, 0);
assert.equal(state.rainM3, 0);
assert.equal(state.balanceErrorM3, 0);
assert.deepEqual(state.plots.map(plot => [plot.id, plot.areaM2, plot.imperviousShare]), [
  ["plot:sealed", 100, 0.8], ["plot:green", 100, 0.2],
]);
assert.equal(state.history.length, 1);
instance.advance(0);
assert.deepEqual(instance.read(), state);
