import assert from "node:assert/strict";
import { model } from "../../../models/city-stormwater/model.ts";

const instance = model.create();
instance.advance();
let state = instance.read();
const close = (actual: number, expected: number) => assert.ok(Math.abs(actual - expected) < 1e-10, `${actual} != ${expected}`);
assert.equal(state.tick, 1);
close(state.rainM3, 4);
close(state.infiltratedM3, 1);
close(state.runoffM3, 3);
close(state.storedM3, 2);
close(state.dischargedM3, 0.5);
close(state.overflowM3, 0.5);
close(state.plots[0]!.intervalRunoffM3, 1.8);
close(state.plots[1]!.intervalRunoffM3, 1.2);
instance.advance();
state = instance.read();
close(state.rainM3, 8);
close(state.infiltratedM3, 2);
close(state.runoffM3, 6);
close(state.storedM3, 2);
close(state.dischargedM3, 1);
close(state.overflowM3, 3);
instance.advance(500);
state = instance.read();
close(state.rainM3, state.infiltratedM3 + state.storedM3 + state.dischargedM3 + state.overflowM3);
close(state.runoffM3, state.storedM3 + state.dischargedM3 + state.overflowM3);
close(state.balanceErrorM3, 0);
assert.equal(state.history.length, 120);
assert.equal(state.history.at(-1)!.tick, 502);
